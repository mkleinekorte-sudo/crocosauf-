static inline void drawDigit3x7(uint8_t d, int x, int y) {
  if (d > 9) return;
  for (int r=0;r<7;r++){
    uint8_t row = DIG3x7[d][r];
    for (int c=0;c<3;c++){
      if (row & (1<<(2-c))) matrix.drawPixel(x+c, y+r, LED_ON);
    }
  }
}

static inline void drawNumberCentered(uint8_t n) {
  const int y = 0;
  matrix.clear();
  if (n <= 9) {
    drawDigit3x7(n, 2, y);
  } else {
    drawDigit3x7(1, 0, y);
    drawDigit3x7(0, 4, y);
  }
  matrix.writeDisplay();
}

static void ensureGameMusicIfNeeded(bool maulOffen) {
  if (!audioAvailable()) return;
  if (welcomeActive) return;
  if (isMuted) return;
  if (!maulOffen) return;

  if (millis() - lastGameMusicCmdMs < 1100) return;

  if (spielLaeuft && !isGameOver && !neueRundePhase && !reminderRunning) {
    if (dfIsBusyPlaying()) { dfIdleSinceMs = 0; return; }

    if (dfIdleSinceMs == 0) dfIdleSinceMs = millis();
    if (millis() - dfIdleSinceMs < 800) return;

    startLoopGameMusicNow();
  }
}

static bool readDebouncedReedOpen() {
  bool raw = (digitalRead(MAUL_SENSOR_PIN) == HIGH);
  unsigned long now = millis();
  if (raw != maulOffenLastRaw) {
    maulOffenLastRaw = raw;
    maulLastChange = now;
  }
  if (now - maulLastChange >= REED_DEBOUNCE_MS) maulOffenStable = raw;
  return maulOffenStable;
}

static bool readDebouncedButton() {
  static bool lastRaw = HIGH, stable = HIGH;
  static unsigned long lastChange = 0;
  bool raw = digitalRead(VOLUME_TASTER);
  unsigned long now = millis();
  if (raw != lastRaw) { lastRaw = raw; lastChange = now; }
  if (now - lastChange >= BUTTON_DEBOUNCE_MS) stable = raw;
  return stable;
}

static int pickFromMask_1based(int count, uint32_t mask, bool randomMode, int &lastIdx) {
  if (count <= 0) return 1;

  uint32_t validMask = mask;
  if (count < 32) validMask &= ((1UL << count) - 1UL);
  if (validMask == 0) validMask = (count >= 32) ? 0xFFFFFFFFUL : ((1UL << count) - 1UL);

  if (!randomMode) {
    for (int step = 1; step <= count; step++) {
      int idx = ((lastIdx + step - 1) % count) + 1;
      if (validMask & (1UL << (idx - 1))) { lastIdx = idx; return idx; }
    }
    lastIdx = 1; return 1;
  }

  int candidates[64]; int cnt = 0;
  for (int i = 1; i <= count; i++) if (validMask & (1UL << (i - 1))) candidates[cnt++] = i;
  if (cnt == 0) { lastIdx = 1; return 1; }

  if (cnt > 1) {
    int filtered[64]; int fcnt = 0;
    for (int i = 0; i < cnt; i++) if (candidates[i] != lastIdx) filtered[fcnt++] = candidates[i];
    int idx = filtered[random(fcnt)];
    lastIdx = idx; return idx;
  }

  lastIdx = candidates[0];
  return candidates[0];
}

static int pickEffectFromMask_0based(int count, uint32_t mask, bool randomMode, int &lastId) {
  if (count <= 0) return 0;

  uint32_t validMask = mask;
  if (count < 32) validMask &= ((1UL << count) - 1UL);
  if (validMask == 0) validMask = (count >= 32) ? 0xFFFFFFFFUL : ((1UL << count) - 1UL);

  if (!randomMode) {
    for (int step = 1; step <= count; step++) {
      int id = (lastId + step) % count;
      if (validMask & (1UL << id)) { lastId = id; return id; }
    }
    lastId = 0; return 0;
  }

  int candidates[64]; int cnt = 0;
  for (int i = 0; i < count; i++) if (validMask & (1UL << i)) candidates[cnt++] = i;
  if (cnt == 0) { lastId = 0; return 0; }

  if (cnt > 1) {
    int filtered[64]; int fcnt = 0;
    for (int i = 0; i < cnt; i++) if (candidates[i] != lastId) filtered[fcnt++] = candidates[i];
    int id = filtered[random(fcnt)];
    lastId = id; return id;
  }

  lastId = candidates[0];
  return candidates[0];
}

static void dispRender() {
  if (dispUntil != 0 && millis() > dispUntil) {
    dispUntil = 0;

    if (reminderRunning) {
      if (reminderShowCup) dispSetCupAnim(255, 0);
      else dispSetReminderAnim();
      return;
    }

    if (isGameOver && (digitalRead(MODUS_PIN) != LOW)) {
      dispSetScroll(standbyText);
      return;
    }

    if (digitalRead(MODUS_PIN) == LOW) { dispSetPinchBlink(); return; }
    if (spielLaeuft) { dispSetGameAnim(); return; }
    dispSetScroll(standbyText);
    return;
  }

  if (millis() - dispLast < 70) return;
  dispLast = millis();

  if (dispMode == DISP_PINCH_BLINK) {
    unsigned long now = millis();
    if (pinchBlinkTogglesLeft > 0) {
      if (now - pinchBlinkLast >= PINCH_NUM_BLINK_MS) {
        pinchBlinkLast = now;
        pinchBlinkOn = !pinchBlinkOn;
        pinchBlinkTogglesLeft--;
        if (pinchBlinkTogglesLeft == 0) pinchBlinkOn = true;
      }
      if (pinchBlinkOn) drawNumberCentered((uint8_t)displayedPinchNumber());
      else dispHardClear();
    } else {
      drawNumberCentered((uint8_t)displayedPinchNumber());
    }
    return;
  }

  if (dispMode == DISP_GAME_ANIM) {
    const unsigned long SWITCH_MS = 1000;
    const unsigned long FRAME_MS  = 110;
    unsigned long now = millis();

    if (gameShowLastSwitch == 0) gameShowLastSwitch = now;
    if (gameFrameLast == 0) gameFrameLast = now;

    if (forcedGameShow < 0) {
      if (now - gameShowLastSwitch >= SWITCH_MS) {
        gameShowLastSwitch = now;
        gameShowId = (uint8_t)pickEffectFromMask_0based(DISP_ANIM_COUNT, (uint32_t)gameDispMask, gameDispRandom, lastGameShowPick);
        gameFrame = 0;
        gameFrameLast = now;
      }
    } else {
      gameShowId = (uint8_t)forcedGameShow;
    }

    if (now - gameFrameLast >= FRAME_MS) { gameFrameLast = now; gameFrame++; }

    matrix.clear();
    auto drawSmiley = [&](bool wink, bool bigSmile){
      matrix.drawPixel(2,2,LED_ON);
      matrix.drawPixel(5,2,LED_ON);
      if (wink) { matrix.drawPixel(5,2,LED_OFF); matrix.drawPixel(5,3,LED_ON); }
      if (bigSmile) {
        matrix.drawPixel(2,5,LED_ON); matrix.drawPixel(3,6,LED_ON);
        matrix.drawPixel(4,6,LED_ON); matrix.drawPixel(5,5,LED_ON);
      } else {
        matrix.drawLine(2,6,5,6,LED_ON);
      }
    };

    switch (gameShowId) {
      case 0: {
        bool wink = ((gameFrame % 10) == 3 || (gameFrame % 10) == 4);
        bool mouthOpen = ((gameFrame % 12) >= 6);
        matrix.drawPixel(2,2,LED_ON); matrix.drawPixel(2,3,LED_ON);
        if (!wink) { matrix.drawPixel(5,2,LED_ON); matrix.drawPixel(5,3,LED_ON); }
        else { matrix.drawPixel(5,3,LED_ON); }

    matrix.writeDisplay();
    return;
  }
}

static void updateWelcomeState() {
  if (!welcomeActive) return;
  unsigned long now = millis();

  if (!welcomeSawBusyLow) {
    if (dfIsBusyPlaying()) welcomeSawBusyLow = true;
    return;
  }

  if (!dfIsBusyPlaying() && (now - welcomeStartedMs) >= MIN_WELCOME_MS) {
    welcomeActive = false;
    applyCurrentGameVolume();
  }
}

// ============================================================
// Volume/mute
// ============================================================
static void handleVolumeAndMute(bool maulOffen) {
  if (!maulOffen) return;

  bool buttonState = readDebouncedButton();
  static bool lastStable = HIGH;

  if (buttonState == LOW && lastStable == HIGH) { buttonPressStart = millis(); buttonHeld = false; }

  if (buttonState == LOW && !buttonHeld && (millis() - buttonPressStart >= muteThreshold)) {
    buttonHeld = true;
    isMuted = !isMuted;

    if (isMuted) {
      lastVolumeBeforeMute = max(volumeLevel, volumeMin);
      volumeLevel = 0;
      applyCurrentGameVolume();
      ensureGameMusicIfNeeded(maulOffen);
      showMutedOnMatrix();
    } else {
      volumeLevel = lastVolumeBeforeMute;
      isMuted = false;
      applyCurrentGameVolume();
      showVolumeOnMatrix();
    }
    prefs.putInt("vol", volumeLevel);
    prefs.putBool("mut", isMuted);
    lastUserActivityMs = millis();
  }

  if (buttonState == HIGH && lastStable == LOW && !buttonHeld) {
    int v = volumeLevel;
    if (isMuted) v = lastVolumeBeforeMute;
    v += 5;
    if (v > volumeMax) v = volumeMin;
    volumeLevel = v;
    isMuted = false;
    applyCurrentGameVolume();
    ensureGameMusicIfNeeded(maulOffen);
    showVolumeOnMatrix();
    prefs.putInt("vol", volumeLevel);
    prefs.putBool("mut", isMuted);
    lastUserActivityMs = millis();
  }

  lastStable = buttonState;
}

// ============================================================
// LED Effects
// ============================================================
static void resetEffects() { effectPos = 0; direction = 1; hue = 0; brightness = 0; brightnessDelta = 5; }

static void effect_render(int id) {
  if (millis() - lastEffectUpdate < 30) return;
  lastEffectUpdate = millis();

  switch (id) { case 10: case 13: case 15: case 16: case 17: case 18: break; default: FastLED.clear(); break; }

  switch (id) {
    case 0: { leds[effectPos] = CRGB::Blue; effectPos += direction; if (effectPos == NUM_LEDS - 1 || effectPos == 0) direction = -direction; break; }
    case 1: leds[random(NUM_LEDS)] = CRGB::White; break;
    case 2: { brightness += brightnessDelta; if (brightness <= 0 || brightness >= 255) brightnessDelta = -brightnessDelta; fill_solid(leds, NUM_LEDS, CRGB(0, brightness, 0)); break; }
    case 3: { brightness += brightnessDelta; if (brightness <= 0 || brightness >= 255) brightnessDelta = -brightnessDelta; fill_solid(leds, NUM_LEDS, CRGB(0, 0, brightness)); break; }
    case 4: { for (int i = 0; i < NUM_LEDS; i++) leds[(effectPos + i) % NUM_LEDS] = CHSV(i * (255 / NUM_LEDS), 255, 255); effectPos = (effectPos + 1) % NUM_LEDS; break; }
    case 5: { for (int i = 0; i < NUM_LEDS; i++) leds[i] = CHSV((i * 20 + hue) % 255, 255, 255); hue++; break; }
    case 6: for (int i = 0; i < NUM_LEDS; i++) leds[i] = CHSV(0, 0, random8(80, 255)); break;
    case 7: leds[random(NUM_LEDS)] = CRGB::White; leds[random(NUM_LEDS)] = CRGB::White; break;
    case 8: { static int b=0; leds[b]=CRGB::Yellow; leds[NUM_LEDS-1-b]=CRGB::Yellow; b=(b+1)%NUM_LEDS; break; }
    case 9: { static int f=0; for(int j=0;j<=f && j<NUM_LEDS;j++) leds[j]=CRGB::Aqua; f=(f+1)%(NUM_LEDS+1); break; }
    case 10:{ fadeToBlackBy(leds,NUM_LEDS,50); static int m=0; leds[m]=CRGB::White; m=(m+1)%NUM_LEDS; break; }
    case 11:{ static uint8_t s=0; s+=2; for(int i=0;i<NUM_LEDS;i++) leds[i]=CHSV(s+i*10,255,255); break; }
    case 12:{ static uint8_t fx=0; fx+=2; for(int i=0;i<NUM_LEDS;i++) leds[i]=CHSV(fx+i*12,255,sin8((millis()/5)+i*20)); break; }
    case 13:{ fadeToBlackBy(leds,NUM_LEDS,60); static int p=0; static uint8_t sh=0; sh+=4; leds[p]=CHSV(sh,255,255); p=(p+1)%NUM_LEDS; break; }
    case 14:{ static int u=0; leds[u]=CRGB::Cyan; u=(u+1)%NUM_LEDS; break; }
    case 15:{ fadeToBlackBy(leds,NUM_LEDS,20); int pos=random(NUM_LEDS); leds[pos]+=CHSV(random8(),200,255); break; }
    case 16:{ fadeToBlackBy(leds,NUM_LEDS,80); static int t=0; for(int i=0;i<NUM_LEDS;i+=3){int idx=(i+t)%NUM_LEDS; leds[idx]=CRGB::White;} t=(t+1)%3; break; }
    case 17:{ static uint8_t base=0; base+=4; for(int i=0;i<NUM_LEDS;i++){ uint8_t wave=sin8((uint8_t)(base+i*18)); leds[i]=CHSV((uint8_t)(base+i*9),255,wave);} if(random8()<120) leds[random(NUM_LEDS)]=CRGB::White; if(random8()<80) leds[random(NUM_LEDS)]+=CHSV(random8(),200,255); break; }
    case 18:{ fadeToBlackBy(leds,NUM_LEDS,40); static uint8_t base=0; base+=2; int pos=random(NUM_LEDS); leds[pos]=CHSV(base+random8(64),255,255); break; }
    case 19:{ static bool flip=false; flip=!flip; fill_solid(leds,NUM_LEDS, flip?CRGB::Red:CRGB::Blue); break; }
  }

  FastLED.show();
}

// Pinchen / GameOver blink
// ============================================================
static int findeNaechstesPinnchen() {
  while (true) {
    int idx = random(PINCHEN_TOTAL);
    if (!pinnchenVerwendet[idx]) return idx;
  }
}

static void blinkGameOver(bool pinchenAktiv) {
  unsigned long now = millis();
  if (now - lastGameOverBlink > PINCHEN_BLINK_INTERVAL) {
    blinkToggle = !blinkToggle;
    lastGameOverBlink = now;
  }
  FastLED.clear();

  if (now - gameOverStart < RED_PHASE_DURATION) {
    if (blinkToggle) fill_solid(leds, NUM_LEDS, CRGB::Red);
  } else if (pinchenAktiv && (now - gameOverStart < RED_PHASE_DURATION + PINCHEN_PHASE_DURATION)) {
    if (!pinnchenGezeigt) {
      pinnchenIndex = findeNaechstesPinnchen();
      pinnchenVerwendet[pinnchenIndex] = true;
      pinnchenGezeigt = true;
    }
    if (blinkToggle) {
      int a = pinnchenLedA[pinnchenIndex];
      int b = pinnchenLedB[pinnchenIndex];
      if (a >= 0 && a < NUM_LEDS) leds[a] = CRGB::Cyan;
      if (b >= 0 && b < NUM_LEDS) leds[b] = CRGB::Cyan;
    }
  }
  FastLED.show();
}

// ============================================================
// DeepSleep
// ============================================================
static void enterDeepSleepNow() {
  if (digitalRead(MAUL_SENSOR_PIN) == HIGH) return;
  if (bleConnected) return;

  reminderStop();
  uiStopAll();
  if (audioAvailable() && !welcomeActive) { player.stop(); delay(120); }

  if (bleStarted) {
    BLEDevice::deinit(true);
    bleStarted = false;
    bleConnected = false;
    bleApproved = false;
  }

  rtc_gpio_deinit((gpio_num_t)MAUL_SENSOR_PIN);
  rtc_gpio_set_direction((gpio_num_t)MAUL_SENSOR_PIN, RTC_GPIO_MODE_INPUT_ONLY);
  rtc_gpio_pullup_en((gpio_num_t)MAUL_SENSOR_PIN);
  rtc_gpio_pulldown_dis((gpio_num_t)MAUL_SENSOR_PIN);

  esp_sleep_enable_ext0_wakeup((gpio_num_t)MAUL_SENSOR_PIN, 1);
  if (reminderEnabled) esp_sleep_enable_timer_wakeup((uint64_t)reminderIntervalMs * 1000ULL);

  delay(50);
  esp_deep_sleep_start();
}

static void startLoopGameMusicNow() {
  if (!audioAvailable()) return;
  if (isMuted) return;

  if (welcomeActive) { pendingResumeMusicAfterWelcome = true; return; }

  unsigned long now = millis();
  if (now - lastGameMusicCmdMs < 900) return;

  if (currentGameMusicTrack <= 0) {
    static int lastGame = 1;
    currentGameMusicTrack = pickFromMask_1based(ROOT_GAME_TRACK_COUNT, gameTrackMask, gameTracksRandom, lastGame);
  }

  lastGameMusicCmdMs = now;
  dfIdleSinceMs = 0;

  player.stop(); delay(180);
  applyCurrentGameVolume(); delay(120);
  player.loop(currentGameMusicTrack);
}

// ============================================================
// Game start
// ============================================================
static void startGameNow(bool pinchenModus) {
  if (welcomeActive) {
    pendingStartAfterWelcome = true;
    pendingStartPinchenMode = pinchenModus;
    return;
  }

  spielLaeuft = true;
  gameReady   = false;
  lastUserActivityMs = millis();

  currentGameMusicTrack = 0;
  dfIdleSinceMs = 0;

  uiPlaying = false; uiMode = 0; uiCurrentTrack = 0;
  uiEffectTesting = false; uiEffectId = -1;
  uiDispTesting = false;   uiDispId = -1;
  forcedGameShow = -1;

  if (pinchenModus) dispSetPinchBlink();
  else dispSetGameAnim();

  startLoopGameMusicNow();

  resetEffects();
  lastEffectSwitch = millis();
  static int lastGE = 0;
  currentEffect = pickEffectFromMask_0based(EFFECT_COUNT, gameEffectMask, gameEffectsRandom, lastGE);

  FastLED.clear(); FastLED.show();
}

static void tryAutoStartIfAlreadyOpen(bool pinchenModus, bool maulOffen) {
  if (!maulOffen) { openCandidate = false; return; }
  if (!gameReady || spielLaeuft || isGameOver || neueRundePhase || reminderRunning) return;

  if (welcomeActive) { pendingStartAfterWelcome = true; pendingStartPinchenMode = pinchenModus; return; }

  if (millis() - maulOpenSince >= OPEN_CONFIRM_MS) {
    openCandidate = false;
    startGameNow(pinchenModus);
  } else {
    openCandidate = true;
    openSince = maulOpenSince;
  }
}

// ============================================================
// MODE SWITCH
// ============================================================
static void switchToPinchenMode(bool maulOffen) {
  dispHardClear();
  for (int i = 0; i < PINCHEN_TOTAL; i++) pinnchenVerwendet[i] = false;

  pinchRoundsDone = 0;
  rundeAktuell   = 1;

  isGameOver     = false;
  spielLaeuft    = false;
  gameReady      = true;
  neueRundePhase = false;
  openCandidate  = false;

  reminderStop();
  uiStopAll();
  FastLED.clear(); FastLED.show();

  dispSetPinchBlink();
  lastUserActivityMs = millis();

  tryAutoStartIfAlreadyOpen(true, maulOffen);
}

static void switchToNormalMode(bool maulOffen) {
  for (int i = 0; i < PINCHEN_TOTAL; i++) pinnchenVerwendet[i] = false;

  pinchRoundsDone = 0;
  rundeAktuell = 1;

  pinnchenGezeigt = false;
  neueRundePhase = false;

  uiStopAll();
  reminderStop();

  if (isGameOver) {
    dispSetCupAnim(255, NORMAL_GAMEOVER_CUP_MS);
  } else if (spielLaeuft && maulOffen) {
    dispSetGameAnim();
    startLoopGameMusicNow();
  } else {
    FastLED.clear(); FastLED.show();
    dispHardClear();
    dispSetScroll(standbyText);
  }

  lastUserActivityMs = millis();
  tryAutoStartIfAlreadyOpen(false, maulOffen);
}

// ============================================================
// BLE APP API
// Protocol: "ID /path?params\n"  ->  "ID HTTP_CODE body\n"
// ============================================================
static String jsonEscape(const String& in) {
  String out; out.reserve(in.length() + 8);
  for (size_t i=0; i<in.length(); i++) {
    char c = in[i];
    if (c == '\\' || c == '"') { out += '\\'; out += c; }
    else if (c == '\n') out += "\\n";
    else if (c == '\r') out += "\\r";
    else if ((uint8_t)c >= 0x20) out += c;
  }
  return out;
}

static int hexNibble(char c) {
  if (c >= '0' && c <= '9') return c - '0';
  if (c >= 'a' && c <= 'f') return c - 'a' + 10;
  if (c >= 'A' && c <= 'F') return c - 'A' + 10;
  return -1;
}

static String urlDecode(const String& in) {
  String out; out.reserve(in.length());
  for (size_t i=0; i<in.length(); i++) {
    char c = in[i];
    if (c == '+') out += ' ';
    else if (c == '%' && i + 2 < in.length()) {
      int a = hexNibble(in[i+1]);
      int b = hexNibble(in[i+2]);
      if (a >= 0 && b >= 0) { out += (char)((a << 4) | b); i += 2; }
      else out += c;
    } else out += c;
  }
  return out;
}

static String queryArg(const String& query, const String& key, const String& def = "") {
  int pos = 0;
  while (pos <= (int)query.length()) {
    int amp = query.indexOf('&', pos);
    if (amp < 0) amp = query.length();
    String part = query.substring(pos, amp);
    int eq = part.indexOf('=');
    String k = (eq >= 0) ? part.substring(0, eq) : part;
    String v = (eq >= 0) ? part.substring(eq + 1) : "";
    if (k == key) return urlDecode(v);
    pos = amp + 1;
    if (amp >= (int)query.length()) break;
  }
  return def;
}

static bool queryHas(const String& query, const String& key) {
  int pos = 0;
  while (pos <= (int)query.length()) {
    int amp = query.indexOf('&', pos);
    if (amp < 0) amp = query.length();
    String part = query.substring(pos, amp);
    int eq = part.indexOf('=');
    String k = (eq >= 0) ? part.substring(0, eq) : part;
    if (k == key) return true;
    pos = amp + 1;
    if (amp >= (int)query.length()) break;
  }
  return false;
}

static bool argTruthy(const String& query, const String& key) {
  if (!queryHas(query, key)) return false;
  String v = queryArg(query, key, "1");
  v.toLowerCase();
  return !(v == "0" || v == "false" || v == "off" || v == "no");
}

static uint32_t maskFromQuery(const String& query, int count) {
  uint32_t m = 0;
  for (int i=0; i<count && i<32; i++) {
    String key = "m" + String(i);
    if (queryHas(query, key) && argTruthy(query, key)) m |= (1UL << i);
  }
  return m;
}

static void bleNotifyRaw(const String& msg) {
  if (!bleConnected || bleTx == nullptr) return;
  const int CHUNK = 20; // kompatibel ohne MTU-Aushandlung
  for (int pos=0; pos<(int)msg.length(); pos += CHUNK) {
    String part = msg.substring(pos, min(pos + CHUNK, (int)msg.length()));
    bleTx->setValue((uint8_t*)part.c_str(), part.length());
    bleTx->notify();
    delay(10);
  }
}

static void bleReply(const String& id, int code, const String& body) {
  bleNotifyRaw(id + " " + String(code) + " " + body + "\n");
}

static bool bleQueuePush(const String& line) {
  uint8_t next = (uint8_t)((bleQHead + 1) % BLE_QUEUE_SIZE);
  if (next == bleQTail) return false;
  bleQueue[bleQHead] = line;
  bleQHead = next;
  return true;
}

static bool bleQueuePop(String& line) {
  if (bleQTail == bleQHead) return false;
  line = bleQueue[bleQTail];
  bleQueue[bleQTail] = "";
  bleQTail = (uint8_t)((bleQTail + 1) % BLE_QUEUE_SIZE);
  return true;
}

class CrocoServerCallbacks : public BLEServerCallbacks {
  void onConnect(BLEServer* pServer) override {
    bleConnected = true;
    bleApproved = false;
    bleConnectedAt = millis();
    bleApproveTracking = false;
    bleButtonArmed = false;
    bleRxAccum = "";
    bleQHead = bleQTail = 0;
  }
  void onDisconnect(BLEServer* pServer) override {
    bleConnected = false;
    bleApproved = false;
    bleApproveTracking = false;
    bleButtonArmed = false;
    bleRxAccum = "";
    bleQHead = bleQTail = 0;
    delay(30);
    BLEDevice::startAdvertising();
  }
};

class CrocoRxCallbacks : public BLECharacteristicCallbacks {
  void onWrite(BLECharacteristic* c) override {
    auto raw = c->getValue();
    String v(raw.c_str());
    if (v.length() == 0) return;
    for (size_t i=0; i<v.length(); i++) {
      char ch = v[i];
      if (ch == '\r') continue;
      if (ch == '\n') {
        String line = bleRxAccum;
        bleRxAccum = "";
        line.trim();
        if (line.length() > 0) bleQueuePush(line);
      } else if (bleRxAccum.length() < BLE_MAX_LINE) {
        bleRxAccum += ch;
      } else {
        bleRxAccum = "";
      }
    }
  }
};

static void doVolUp() {
  int v = volumeLevel;
  if (isMuted) v = lastVolumeBeforeMute;
  v += 5;
  if (v > volumeMax) v = volumeMin;
  volumeLevel = v;
  lastVolumeBeforeMute = v;
  isMuted = false;
  applyCurrentGameVolume();
  showVolumeOnMatrix();
  prefs.putInt("vol", volumeLevel);
  prefs.putBool("mut", isMuted);
  lastUserActivityMs = millis();
}

static void doMuteToggle() {
  isMuted = !isMuted;
  if (isMuted) {
    lastVolumeBeforeMute = max(volumeLevel, volumeMin);
    volumeLevel = 0;
    showMutedOnMatrix();
  } else {
    volumeLevel = max(lastVolumeBeforeMute, volumeMin);
    showVolumeOnMatrix();
  }
  applyCurrentGameVolume();
  prefs.putInt("vol", volumeLevel);
  prefs.putBool("mut", isMuted);
  lastUserActivityMs = millis();
}

static String appPlay(const String& query) {
  if (!audioAvailable()) return "AUDIO_OFF";
  if (welcomeActive) return "WELCOME_PLAYING";

  String src = queryArg(query, "src");
  int t = queryArg(query, "t", "0").toInt();
  if (t <= 0) return "BAD_TRACK";

  int newMode = 0;
  if (src == "g" && t <= ROOT_GAME_TRACK_COUNT) newMode = 1;
  else if (src == "w" && t <= WELCOME_TRACK_COUNT) newMode = 2;
  else if (src == "go" && t <= GAMEOVER_TRACK_COUNT) newMode = 3;
  else if (src == "r" && t <= REMINDER_TRACK_COUNT) newMode = 4;
  else if (src == "nr" && t <= NEWRND_TRACK_COUNT) newMode = 5;
  else return "BAD_SRC_OR_TRACK";

  if (uiPlaying && uiMode == newMode && uiCurrentTrack == t) {
    player.stop(); delay(120);
    uiPlaying = false; uiMode = 0; uiCurrentTrack = 0;
    return "STOP";
  }
  if (gameIsBusyForUi()) return "BUSY";

  uiStopAll();
  applyCurrentGameVolume();
  delay(80);
  if (newMode == 1) player.play(t);
  else if (newMode == 2) player.playFolder(FOLDER_SYSTEM, t);
  else if (newMode == 3) player.playFolder(FOLDER_GAMEOVER, t);
  else if (newMode == 4) player.playFolder(FOLDER_REMINDER, t);
  else if (newMode == 5) player.playFolder(FOLDER_NEWRND, t);
  uiPlaying = true; uiMode = newMode; uiCurrentTrack = t;
  return "PLAY";
}

  if (gameIsBusyForUi()) return "BUSY";
  int id = queryArg(query, "id", "-1").toInt();
  if (id < 0 || id >= EFFECT_COUNT) return "BAD";

    uiEffectTesting = false; uiEffectId = -1;

    return "STOP";
  }
  uiStopAll();
  uiEffectTesting = true; uiEffectId = id;
  resetEffects(); lastEffectUpdate = 0;
  return "OK";
}

static String appSaveGroup(const String& query) {
  String group = queryArg(query, "group");
  bool rnd = (queryArg(query, "mode", "seq") == "rnd");

  if (group == "g") {
    uint32_t m = maskFromQuery(query, ROOT_GAME_TRACK_COUNT); if (!m) m = DEFAULT_GAME_MASK;
    gameTrackMask = (uint16_t)m; gameTracksRandom = rnd;
    prefs.putUShort("gMask", gameTrackMask); prefs.putBool("gRnd", gameTracksRandom);
  } else if (group == "go") {
    uint32_t m = maskFromQuery(query, GAMEOVER_TRACK_COUNT); if (!m) m = DEFAULT_GAMEOVER_MASK;
    gameOverMask = (uint16_t)m; gameOverRandom = rnd;
    prefs.putUShort("goMask", gameOverMask); prefs.putBool("goRnd", gameOverRandom);
  } else if (group == "w") {
    uint32_t m = maskFromQuery(query, WELCOME_TRACK_COUNT); if (!m) m = DEFAULT_WELCOME_MASK;
    welcomeMask = (uint8_t)m; welcomeRandom = rnd;
    prefs.putUChar("wMask", welcomeMask); prefs.putBool("wRnd", welcomeRandom);
  } else if (group == "r") {
    uint32_t m = maskFromQuery(query, REMINDER_TRACK_COUNT); if (!m) m = DEFAULT_REMINDER_MASK;
    reminderMask = (uint8_t)m; reminderSoundRandom = rnd;
    prefs.putUChar("rMask", reminderMask); prefs.putBool("rRnd", reminderSoundRandom);
  } else if (group == "nr") {
    uint32_t m = maskFromQuery(query, NEWRND_TRACK_COUNT); if (!m) m = DEFAULT_NEWRND_MASK;
    newRndMask = (uint8_t)m; newRndRandom = rnd;
    prefs.putUChar("nrMask", newRndMask); prefs.putBool("nrRnd", newRndRandom);
  } else if (group == "ge") {
    uint32_t m = maskFromQuery(query, EFFECT_COUNT); if (!m) m = 0x000FFFFF;
    gameEffectMask = m; gameEffectsRandom = rnd;
    prefs.putULong("geMask", gameEffectMask); prefs.putBool("geRnd", gameEffectsRandom);
  } else if (group == "re") {
    uint32_t m = maskFromQuery(query, EFFECT_COUNT);
    if (!m) m = (1UL<<1)|(1UL<<4)|(1UL<<10)|(1UL<<11)|(1UL<<15)|(1UL<<18);
    reminderEffectMask = m; reminderEffectsRandom = rnd;
    prefs.putULong("reMask", reminderEffectMask); prefs.putBool("reRnd", reminderEffectsRandom);
  } else if (group == "gd") {
    uint32_t m = maskFromQuery(query, DISP_ANIM_COUNT); if (!m) m = 0x000FFFFF;
    gameDispMask = m; gameDispRandom = rnd;
    prefs.putULong("gdMask", gameDispMask); prefs.putBool("gdRnd", gameDispRandom);
  } else return "BAD_GROUP";

  lastUserActivityMs = millis();
  return "OK";
}

static void applyDisplayConfig(const String& query) {
  if (queryHas(query, "rot")) {
    displayRotation = (uint8_t)constrain(queryArg(query, "rot", "0").toInt(), 0, 3);
    prefs.putUChar("dRot", displayRotation);
  }
  if (queryHas(query, "lbr")) {
    ledBrightnessPct = (uint8_t)constrain(queryArg(query, "lbr", "100").toInt(), 0, 100);
    prefs.putUChar("lBr", ledBrightnessPct); applyLedBrightness();
  }
  if (queryHas(query, "dbr")) {
    displayBrightnessPct = (uint8_t)constrain(queryArg(query, "dbr", "100").toInt(), 0, 100);
    prefs.putUChar("dBr", displayBrightnessPct); applyDisplayBrightness();
  }
  matrix.setRotation(displayRotation & 3);
  dispHardClear();
  if (digitalRead(MODUS_PIN) == LOW) dispSetPinchBlink(); else dispSetScroll(standbyText);
  lastUserActivityMs = millis();
}

static void applyAudioConfig(const String& query) {
  bool requestedAudio = queryHas(query, "aen") ? argTruthy(query, "aen") : audioEnabled;
  audioEnabled = requestedAudio;
  prefs.putBool("aEn", audioEnabled);

  if (!audioEnabled) {
    welcomeActive = false;
    pendingResumeMusicAfterWelcome = false;
    pendingStartAfterWelcome = false;
    if (dfplayerInitOk) { player.stop(); delay(120); }
  } else {
    initDfPlayerIfNeeded();
  }

  if (queryHas(query, "vstart")) {
    int vStart = constrain(queryArg(query, "vstart", "20").toInt(), 0, 30);
    if (vStart == 0) {
      if (!isMuted) lastVolumeBeforeMute = max(volumeLevel, volumeMin);
      isMuted = true; volumeLevel = 0;
    } else {
      isMuted = false;
      volumeLevel = constrain(vStart, volumeMin, volumeMax);
      lastVolumeBeforeMute = volumeLevel;
    }
  }
  if (queryHas(query, "vwelcome")) {
    welcomeVolumeLevel = constrain(queryArg(query, "vwelcome", "20").toInt(), 0, 30);
  }

  prefs.putInt("vol", volumeLevel);
  prefs.putBool("mut", isMuted);
  prefs.putInt("wVol", welcomeVolumeLevel);
  applyCurrentGameVolume();
  showVolumeOnMatrix();
  lastUserActivityMs = millis();
}

  prefs.putUShort("remMin", remMin);
  prefs.putUShort("remDur", reminderEffectDurationMs);

  lastReminderCheckMs = millis();
}

static void processBleCommand(const String& line) {
  int sp = line.indexOf(' ');
  String id = (sp > 0) ? line.substring(0, sp) : "0";
  String req = (sp > 0) ? line.substring(sp + 1) : line;
  req.trim();

  int qm = req.indexOf('?');
  String path = (qm >= 0) ? req.substring(0, qm) : req;
  String query = (qm >= 0) ? req.substring(qm + 1) : "";

  if (path == "/hello") {
    bleReply(id, 200, buildHelloJson());
    return;
  }
  if (!bleApproved) {
    bleReply(id, 403, "NOT_APPROVED");
    return;
  }

  if (path == "/status") bleReply(id, 200, buildStatusJson());
  else if (path == "/config") bleReply(id, 200, buildConfigJson());
  else if (path == "/stop") {
    if (welcomeActive || spielLaeuft || isGameOver || neueRundePhase) bleReply(id, 409, "BUSY");
    else {
      if (reminderRunning) reminderStop();
      uiStopAll();
      if (digitalRead(MODUS_PIN) == LOW) dispSetPinchBlink(); else dispSetScroll(standbyText);
      lastUserActivityMs = millis();
      bleReply(id, 200, "OK");
    }
  }
  else if (path == "/volUp") { doVolUp(); bleReply(id, 200, "OK"); }
  else if (path == "/muteToggle") { doMuteToggle(); bleReply(id, 200, "OK"); }
  else if (path == "/volume") {
    int v = constrain(queryArg(query, "v", "20").toInt(), volumeMin, volumeMax);
    volumeLevel = v; lastVolumeBeforeMute = v; isMuted = false;
    applyCurrentGameVolume(); showVolumeOnMatrix();
    prefs.putInt("vol", volumeLevel); prefs.putBool("mut", false);
    lastUserActivityMs = millis();
    bleReply(id, 200, "OK");
  }
  else if (path == "/play") bleReply(id, 200, appPlay(query));
  else if (path == "/testEffect") bleReply(id, 200, appTestEffect(query));
  else if (path == "/testDisp") bleReply(id, 200, appTestDisplay(query));
  else if (path == "/save") {
    String r = appSaveGroup(query);
    bleReply(id, r == "OK" ? 200 : 400, r);
  }
  else if (path == "/saveText") {
    String t = queryArg(query, "txt");
    t.trim();
    if (t.length() == 0) t = "CROCOSAUF DELUXE";
    if (t.length() > 40) t = t.substring(0, 40);
    standbyText = t;
    prefs.putString("sTxt", standbyText);
    if (!spielLaeuft && !isGameOver && !neueRundePhase && !reminderRunning && (digitalRead(MODUS_PIN) != LOW)) dispSetScroll(standbyText);
    bleReply(id, 200, "OK");
  }
  else if (path == "/saveReminderCfg") { applyReminderConfig(query); bleReply(id, 200, "OK"); }
  else if (path == "/saveDisplayCfg") { applyDisplayConfig(query); bleReply(id, 200, "OK"); }
  else if (path == "/saveAudioCfg") { applyAudioConfig(query); bleReply(id, 200, "OK"); }
  else if (path == "/showScroll") {
    if (!spielLaeuft && !isGameOver && !neueRundePhase && !reminderRunning) dispSetScroll(standbyText);
    bleReply(id, 200, "OK");
  }
  else if (path == "/testReminder") {
    if (gameIsBusyForUi()) bleReply(id, 409, "BUSY");
    else { reminderStart(); bleReply(id, 200, "OK"); }
  }
  else if (path == "/factoryReset") {
    if (queryArg(query, "confirm") != "yes") bleReply(id, 400, "CONFIRM_REQUIRED");
    else { factoryResetSettings(); bleReply(id, 200, "OK"); }
  }
  else bleReply(id, 404, "NOT_FOUND");
}

static void startBleControl() {
  if (bleStarted) return;
  BLEDevice::init(BLE_DEVICE_NAME);
  bleServer = BLEDevice::createServer();
  bleServer->setCallbacks(new CrocoServerCallbacks());

  BLEService* svc = bleServer->createService(BLE_SERVICE_UUID);
  bleRx = svc->createCharacteristic(BLE_RX_UUID, BLECharacteristic::PROPERTY_WRITE | BLECharacteristic::PROPERTY_WRITE_NR);
  bleTx = svc->createCharacteristic(BLE_TX_UUID, BLECharacteristic::PROPERTY_NOTIFY);
  bleTx->addDescriptor(new BLE2902());
  bleRx->setCallbacks(new CrocoRxCallbacks());
  svc->start();

  BLEAdvertising* adv = BLEDevice::getAdvertising();
  adv->addServiceUUID(BLE_SERVICE_UUID);
  adv->setScanResponse(true);
  adv->setMinPreferred(0x06);
  adv->setMaxPreferred(0x12);
  BLEDevice::startAdvertising();
  bleStarted = true;
}

static void handleBleApproval(bool maulOffen) {
  if (!bleConnected) {
    bleApproveTracking = false;
    bleButtonArmed = false;
    return;
  }

  bool pressed = (readDebouncedButton() == LOW);
  if (!pressed) {
    bleButtonArmed = true;
    bleApproveTracking = false;
  }

  if (bleApproved) return;

  bool readyForApproval = !maulOffen && gameReady && !spielLaeuft && !isGameOver && !neueRundePhase && !reminderRunning && !welcomeActive;
  if (readyForApproval && pressed && bleButtonArmed) {
    if (!bleApproveTracking) {
      bleApproveTracking = true;
      bleApprovePressStart = millis();
    } else if (millis() - bleApprovePressStart >= BLE_APPROVE_HOLD_MS) {
      bleApproved = true;
      bleApproveTracking = false;
      dispSetScroll("APP OK", 1800);
      lastUserActivityMs = millis();
      // Keine unaufgeforderte BLE-Antwort senden: die App pollt /hello mit eigener Request-ID.
    }
  } else if (!readyForApproval) {
    bleApproveTracking = false;
  }
}

static void bleLoop() {
  String line;
  if (bleQueuePop(line)) processBleCommand(line);

  if (bleConnected && !bleApproved && (millis() - bleConnectedAt > BLE_UNAPPROVED_TIMEOUT_MS)) {
    if (bleServer) bleServer->disconnect(bleServer->getConnId());
  }
}

// ============================================================
// Random seed helper
// ============================================================
static inline void seedBetterRandom() {
  uint32_t seed = (uint32_t)micros();
  seed ^= (uint32_t)esp_random();
  randomSeed(seed);
}

// ============================================================
// setup / loop
// ============================================================
void setup() {
  setCpuFrequencyMhz(80);
  Serial.begin(115200);
  delay(150);

  bootCount++;
  esp_sleep_wakeup_cause_t cause = esp_sleep_get_wakeup_cause();
  lastWakeWasTimer = (cause == ESP_SLEEP_WAKEUP_TIMER) ? 1 : 0;

  prefs.begin("crocodoc", false);

  // Load settings
  gameTrackMask = prefs.getUShort("gMask", DEFAULT_GAME_MASK); if (gameTrackMask == 0) gameTrackMask = DEFAULT_GAME_MASK;
  gameOverMask  = prefs.getUShort("goMask", DEFAULT_GAMEOVER_MASK); if (gameOverMask == 0) gameOverMask = DEFAULT_GAMEOVER_MASK;
  welcomeMask   = prefs.getUChar("wMask", DEFAULT_WELCOME_MASK); if (welcomeMask == 0) welcomeMask = DEFAULT_WELCOME_MASK;
  reminderMask  = prefs.getUChar("rMask", DEFAULT_REMINDER_MASK); if (reminderMask == 0) reminderMask = DEFAULT_REMINDER_MASK;
  newRndMask    = prefs.getUChar("nrMask", DEFAULT_NEWRND_MASK); if (newRndMask == 0) newRndMask = DEFAULT_NEWRND_MASK;

  gameTracksRandom    = prefs.getBool("gRnd", true);
  gameOverRandom      = prefs.getBool("goRnd", false);
  welcomeRandom       = prefs.getBool("wRnd", false);
  reminderSoundRandom = prefs.getBool("rRnd", true);
  newRndRandom        = prefs.getBool("nrRnd", false);

  gameEffectMask = prefs.getULong("geMask", 0x000FFFFF);
  if ((gameEffectMask & 0x000FFFFF) == 0) gameEffectMask = 0x000FFFFF;

  reminderEffectMask = prefs.getULong("reMask", (1UL<<1)|(1UL<<4)|(1UL<<10)|(1UL<<11)|(1UL<<15)|(1UL<<18));
  if ((reminderEffectMask & 0x000FFFFF) == 0) reminderEffectMask = (1UL<<1)|(1UL<<4)|(1UL<<10)|(1UL<<11)|(1UL<<15)|(1UL<<18);

  gameEffectsRandom = prefs.getBool("geRnd", false);
  reminderEffectsRandom = prefs.getBool("reRnd", true);

  gameDispMask = prefs.getULong("gdMask", 0x000FFFFF);
  if ((gameDispMask & 0x000FFFFF) == 0) gameDispMask = 0x000FFFFF;
  gameDispRandom = prefs.getBool("gdRnd", false);

  standbyText = prefs.getString("sTxt", "CROCOSAUF DELUXE");
  standbyText.trim();
  if (standbyText.length() == 0) standbyText = "CROCOSAUF DELUXE";
  if (standbyText.length() > 40) standbyText = standbyText.substring(0, 40);

  volumeLevel = prefs.getInt("vol", 20);
  volumeLevel = constrain(volumeLevel, volumeMin, volumeMax);
  isMuted = prefs.getBool("mut", false);
  if (isMuted) volumeLevel = 0;

  welcomeVolumeLevel = prefs.getInt("wVol", 20);
  welcomeVolumeLevel = constrain(welcomeVolumeLevel, 0, 30);

  reminderEnabled      = prefs.getBool("remEn", true);
  reminderSoundEnabled = prefs.getBool("remSnd", true);

  uint16_t remMin = prefs.getUShort("remMin", 8);
  remMin = constrain(remMin, (uint16_t)1, (uint16_t)60);
  reminderIntervalMs = (uint32_t)remMin * 60UL * 1000UL;

  uint16_t rd = prefs.getUShort("remDur", 3500);
  rd = constrain(rd, (uint16_t)1500, (uint16_t)9000);
  reminderEffectDurationMs = rd;

  reminderLedDurationMs = prefs.getUInt("remLed", reminderEffectDurationMs);
  reminderLedDurationMs = constrain(reminderLedDurationMs, 1500UL, 30000UL);

  // NEW load
  displayRotation = prefs.getUChar("dRot", 0) & 3;
  audioEnabled = prefs.getBool("aEn", true);

  // NEW load brightness
  ledBrightnessPct = prefs.getUChar("lBr", 100);
  ledBrightnessPct = constrain(ledBrightnessPct, (uint8_t)0, (uint8_t)100);

  displayBrightnessPct = prefs.getUChar("dBr", 100);
  displayBrightnessPct = constrain(displayBrightnessPct, (uint8_t)0, (uint8_t)100);

  // Pins
  pinMode(MAUL_SENSOR_PIN, INPUT_PULLUP);
  pinMode(VOLUME_TASTER,   INPUT_PULLUP);
  pinMode(MODUS_PIN,       INPUT_PULLUP);
  pinMode(DFPLAYER_BUSY_PIN, INPUT_PULLUP);

  // I2C + Matrix
  Wire.begin(I2C_SDA, I2C_SCL);
  matrix.begin(0x70);
  matrix.setRotation(displayRotation & 3);
  applyDisplayBrightness();
  dispHardClear();

  // LED Strip
  FastLED.addLeds<WS2812B, LED_PIN, GRB>(leds, NUM_LEDS);
  applyLedBrightness();
  FastLED.clear(); FastLED.show();

  // Debounce init
  maulOffenLastRaw = (digitalRead(MAUL_SENSOR_PIN) == HIGH);
  maulOffenStable  = maulOffenLastRaw;
  maulLastChange   = millis();
  maulOpenSince    = maulOffenStable ? millis() : 0;

  // DFPlayer init (FAILSAFE + Audio toggle)
  dfplayerInitOk = false;
  if (audioEnabled) {
    mp3Serial.begin(9600, SERIAL_8N1, DFPLAYER_RX_PIN, DFPLAYER_TX_PIN);
    delay(800);
    for (int i = 0; i < 6 && !dfplayerInitOk; i++) { dfplayerInitOk = player.begin(mp3Serial); delay(300); }
    if (dfplayerInitOk) applyCurrentGameVolume();
  }

  seedBetterRandom();

  lastUserActivityMs = millis();
  lastReminderCheckMs = millis();

  bool coldBootLikely = (cause == ESP_SLEEP_WAKEUP_UNDEFINED);
  if (!lastWakeWasTimer) startBleControl();

  if (digitalRead(MODUS_PIN) == LOW) dispSetPinchBlink();
  else dispSetScroll(standbyText);

  lastPinchenModus = (digitalRead(MODUS_PIN) == LOW);

  // Welcome: kommt immer bei Coldboot, auch wenn vorher Mute gespeichert war
  if (audioAvailable() && coldBootLikely) {
    delay(1200);
    static int lastW = 1;
    int w = pickFromMask_1based(WELCOME_TRACK_COUNT, welcomeMask, welcomeRandom, lastW);

    player.stop(); delay(120);
    applyVolumeToPlayer(welcomeVolumeLevel); delay(120);
    player.playFolder(FOLDER_SYSTEM, w);

    welcomeActive = true;
    welcomeStartedMs = millis();
    welcomeSawBusyLow = false;
  }

  // Wake by timer: Reminder (kurz) dann wieder schlafen
  if (lastWakeWasTimer) {
    bool maulOffen = readDebouncedReedOpen();
    if (!maulOffen) {
      reminderStart();

      while (millis() < reminderUntilMs) {
        dispRender();
        effect_render(reminderEffectId);
        if (readDebouncedReedOpen()) break;
        delay(1);
      }

      reminderStop();
      enterDeepSleepNow();
    } else {
      // Falls der Benutzer genau beim Timer-Wakeup das Maul geoeffnet hat,
      // bleibt das Geraet wach und Bluetooth wird fuer die App gestartet.
      startBleControl();
      lastUserActivityMs = millis();
    }
  }
}

void loop() {
  bleLoop();

  bool maulOffen = readDebouncedReedOpen();

  static bool lastStableOpen = false;
  if (maulOffen && !lastStableOpen) {
    maulOpenSince = millis();
    lastUserActivityMs = millis();
  }
  if (!maulOffen && lastStableOpen) {
    lastUserActivityMs = millis();
  }
  lastStableOpen = maulOffen;

  handleBleApproval(maulOffen);
  handleVolumeAndMute(maulOffen);
  dispRender();

  updateWelcomeState();

  if (!welcomeActive) {
    if (pendingResumeMusicAfterWelcome && spielLaeuft && maulOffen) {
      pendingResumeMusicAfterWelcome = false;
      startLoopGameMusicNow();
    }
    if (pendingStartAfterWelcome) {
      if (gameReady && maulOffen && !spielLaeuft && !isGameOver && !neueRundePhase && !reminderRunning) {
        pendingStartAfterWelcome = false;
        startGameNow(pendingStartPinchenMode);
      }
    }
  }

  // Modusschalter entprellt
  static bool modusLastRaw = false;
  static bool modusStable  = false;
  static unsigned long modusLastChange = 0;

  bool modusRaw = (digitalRead(MODUS_PIN) == LOW);
  unsigned long now = millis();

  if (modusRaw != modusLastRaw) { modusLastRaw = modusRaw; modusLastChange = now; lastUserActivityMs = millis(); }
  if (now - modusLastChange >= MODUS_DEBOUNCE_MS) modusStable = modusRaw;

  bool pinchenModus = modusStable;

  // App-Tests: beim Oeffnen des Mauls sofort beenden, damit das Spiel normal starten kann.
  if (!gameIsBusyForUi()) {
    if (maulOffen && (uiPlaying || uiEffectTesting || uiDispTesting)) {
      uiStopAll();
      if (pinchenModus) dispSetPinchBlink(); else dispSetScroll(standbyText);
    } else {
      if (uiEffectTesting) { effect_render(uiEffectId); delay(1); return; }
      if (uiDispTesting)   { delay(1); return; }
    }
  } else {
    if (uiEffectTesting || uiDispTesting) {
      uiEffectTesting = false; uiEffectId = -1;
      uiDispTesting = false; uiDispId = -1;
      forcedGameShow = -1;
    }
  }

  // MODE SWITCH sofort
  if (pinchenModus != lastPinchenModus) {
    if (pinchenModus) switchToPinchenMode(maulOffen);
    else switchToNormalMode(maulOffen);
    lastPinchenModus = pinchenModus;
  }

  // Reminder running
  if (reminderRunning) {
    if (maulOffen) { reminderStop(); lastUserActivityMs = millis(); }
    else {
      if (millis() - reminderSwapLast >= REMINDER_SWAP_MS) {
        reminderSwapLast = millis();
        reminderShowCup = !reminderShowCup;
        if (reminderShowCup) dispSetCupAnim(255, 0);
        else dispSetReminderAnim();
      }
      effect_render(reminderEffectId);
      if (millis() > reminderUntilMs) { reminderStop(); }
      delay(1);
      return;
    }
  }

  // NEUE RUNDE PHASE
  if (neueRundePhase) {
    openCandidate = false;
    if (millis() - neueRundeStart < GRUEN_PHASE_DURATION) {
      if (millis() - lastGameOverBlink > PINCHEN_BLINK_INTERVAL) {
        blinkToggle = !blinkToggle;
        lastGameOverBlink = millis();
      }
      fill_solid(leds, NUM_LEDS, blinkToggle ? CRGB::Green : CRGB::Black);
      FastLED.show();

      if (dispMode != DISP_SCROLL || dispText != "NEUE RUNDE") {
        dispSetScroll("NEUE RUNDE", GRUEN_PHASE_DURATION + 1000);
      }
    } else {
      neueRundePhase = false;
      gameReady = true;
      FastLED.clear(); FastLED.show();

      pinchRoundsDone = 0;
      rundeAktuell = 1;
      for (int i = 0; i < PINCHEN_TOTAL; i++) pinnchenVerwendet[i] = false;

      dispSetPinchBlink();
      tryAutoStartIfAlreadyOpen(true, maulOffen);
    }
    delay(1);
    return;
  }

  // GAMEOVER PHASE
  if (isGameOver) {
    openCandidate = false;

    if (pinchenModus) {
      if (dispMode != DISP_CUP_ANIM) dispSetCupAnim(255, 0);
    } else {
      if (dispMode != DISP_CUP_ANIM) dispSetCupAnim(255, NORMAL_GAMEOVER_CUP_MS);
    }

    blinkGameOver(pinchenModus);

    unsigned long elapsed = millis() - gameOverStart;
    unsigned long waitTime = RED_PHASE_DURATION + (pinchenModus ? PINCHEN_PHASE_DURATION : 0);

    if (elapsed > waitTime) {
      isGameOver = false;
      pinnchenGezeigt = false;
      FastLED.clear(); FastLED.show();

      if (pinchenModus) {
        pinchenAdvanceAfterGameOver(maulOffen);
      } else {
        gameReady = true;
        lastUserActivityMs = millis();
        tryAutoStartIfAlreadyOpen(false, maulOffen);
      }
    }

    delay(1);
    return;
  }

  // Start-Arming
  if (maulOffen && gameReady && !spielLaeuft) {
    if (!openCandidate) { openCandidate = true; openSince = maulOpenSince; }
    else {
      if (millis() - openSince >= OPEN_CONFIRM_MS) {
        openCandidate = false;
        startGameNow(pinchenModus);
      }
    }
  }
  if (!maulOffen && openCandidate && !spielLaeuft) openCandidate = false;

  // Spiel läuft
  if (spielLaeuft && maulOffen) {
    static unsigned long musicKeepaliveLast = 0;
    if (millis() - musicKeepaliveLast > 2000) {
      musicKeepaliveLast = millis();
      ensureGameMusicIfNeeded(maulOffen);
    }

    if (millis() - lastEffectSwitch > effectInterval) {
      static int lastGE2 = 0;
      currentEffect = pickEffectFromMask_0based(EFFECT_COUNT, gameEffectMask, gameEffectsRandom, lastGE2);
      resetEffects();
      lastEffectSwitch = millis();
    }
    effect_render(currentEffect);
  }

  // Maul zu -> GameOver
  if (!maulOffen && spielLaeuft && !isGameOver) {
    openCandidate = false;
    lastUserActivityMs = millis();

    currentGameMusicTrack = 0;
    dfIdleSinceMs = 0;

    if (audioAvailable() && !welcomeActive && !isMuted) {
      static int lastGO = 1;
      int go = pickFromMask_1based(GAMEOVER_TRACK_COUNT, gameOverMask, gameOverRandom, lastGO);

      player.stop(); delay(120);
      applyCurrentGameVolume(); delay(120);
      player.playFolder(FOLDER_GAMEOVER, go);
    }

    if (pinchenModus) dispSetCupAnim(255, 0);
    else dispSetCupAnim(255, NORMAL_GAMEOVER_CUP_MS);

    isGameOver = true;
    spielLaeuft = false;
    pinnchenGezeigt = false;
    gameReady = false;
    gameOverStart = millis();

    FastLED.clear();
    FastLED.show();
  }

  // Idle -> Reminder + Sleep
  bool idleState = (!spielLaeuft && !isGameOver && !neueRundePhase && gameReady);

  if (idleState) {
    if (reminderEnabled && !maulOffen) {
      if (millis() - lastReminderCheckMs >= reminderIntervalMs) {
        reminderStart();
        lastReminderCheckMs = millis();
      }
    } else {
      lastReminderCheckMs = millis();
    }

    if (!maulOffen && !bleConnected && (millis() - lastUserActivityMs >= DEEP_SLEEP_AFTER_MS)) {
      enterDeepSleepNow();
    }

    delay(10);
  } else {
    delay(1);
  }
}
